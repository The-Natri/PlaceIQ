SQL> -- =============================================================================
SQL> -- SECTION 3: TRIGGER — auto-update placement_status on Offer insert
SQL> -- Fires the moment an offer row is created (i.e. the company has selected
SQL> -- the student), independent of whether the student later accepts/declines
SQL> -- it. This is a deliberate reading of "record offers -> student is Placed":
SQL> -- the offer-acceptance transaction (in the backend) governs Offer.status
SQL> -- (Pending/Accepted/Declined), while this trigger governs the coarser
SQL> -- Student.placement_status (Not Placed/Placed).
SQL> --
SQL> -- WHAT CHANGED FROM db/schema.sql:
SQL> --   - Postgres needs a standalone trigger FUNCTION plus a CREATE TRIGGER
SQL> --     that references it by name. Oracle triggers are self-contained PL/SQL
SQL> --     blocks — the logic goes directly in the CREATE TRIGGER body, no
SQL> --     separate function object exists.
SQL> --   - Row values are referenced as `:NEW.col` (with the leading colon) in
SQL> --     the trigger body — Postgres uses bare `NEW.col`.
SQL> --   - No `LANGUAGE plpgsql` / `$$` delimiters; Oracle just needs `BEGIN ...
SQL> --     END;` followed by a `/` for SQL*Plus.
SQL> --   - Postgres skips offers inserted already Declined with an IF ... RETURN
SQL> --     inside the function; here the same guard is the trigger's WHEN clause,
SQL> --     so the trigger body doesn't even run for them.
SQL> -- =============================================================================
SQL> 
SQL> CREATE OR REPLACE TRIGGER trg_offer_after_insert
  2  AFTER INSERT ON offer
  3  FOR EACH ROW
  4  WHEN (NEW.status <> 'Declined')
  5  BEGIN
  6  	 UPDATE student
  7  	    SET placement_status = 'Placed'
  8  	  WHERE student_id = (
  9  	     SELECT student_id FROM application WHERE application_id = :NEW.application_id
 10  	  );
 11  END;
 12  /

Trigger created.

SQL> -- =============================================================================
SQL> -- SECTION 4: STORED FUNCTIONS — eligibility check
SQL> -- Implemented in the DATABASE (not reimplemented as Python if/else) so the
SQL> -- eligibility rule is enforced identically no matter which client calls it
SQL> -- (Flask backend today, a future script, or a DBA running ad-hoc SQL) and so
SQL> -- it lives in exactly one place to change.
SQL> --
SQL> -- WHAT CHANGED FROM db/schema.sql — this is the biggest semantic rewrite in
SQL> -- the whole port, not just a syntax swap:
SQL> --   1. RETURN TYPE: Postgres returns BOOLEAN and the backend/seed script
SQL> --	calls it straight from SQL (`SELECT fn_check_eligibility(...)`).
SQL> --	Oracle's BOOLEAN type cannot be used in a SQL context (only inside
SQL> --	PL/SQL) before Oracle 23c, so a SQL-callable Oracle version must
SQL> --      return NUMBER instead — 1 means eligible, 0 means not eligible.
SQL> --   2. "ROW NOT FOUND" HANDLING: PL/pgSQL's `SELECT ... INTO` on a missing
SQL> --	row just leaves the variables NULL and sets a `FOUND` flag you check
SQL> --	afterward (`IF NOT FOUND THEN RETURN FALSE; END IF;` in the Postgres
SQL> --	version). Oracle's `SELECT ... INTO` on a missing row instead RAISES
SQL> --      the NO_DATA_FOUND exception immediately — there's no flag to poll.
SQL> --	This function therefore needs an EXCEPTION block, which the Postgres
SQL> --	version doesn't. Whether the student lookup or the drive lookup is
SQL> --	the one that fails, control jumps straight to WHEN NO_DATA_FOUND and
SQL> --      the function returns 0 (not eligible) — the same end result as the
SQL> --	two separate `IF NOT FOUND` checks in the Postgres version, just
SQL> --	reached through Oracle's exception-based control flow instead of a
SQL> --	checked flag.

SQL> CREATE OR REPLACE FUNCTION fn_check_eligibility(
  2  	 p_student_id IN student.student_id%TYPE,
  3  	 p_drive_id   IN drive.drive_id%TYPE
  4  ) RETURN NUMBER
  5  IS
  6  	 v_cgpa 	     student.cgpa%TYPE;
  7  	 v_backlogs	     student.backlogs%TYPE;
  8  	 v_dept_id	     student.dept_id%TYPE;
  9  	 v_min_cgpa	     drive.min_cgpa%TYPE;
 10  	 v_max_backlogs      drive.max_backlogs%TYPE;
 11  	 v_restricted_count  NUMBER;
 12  	 v_dept_match_count  NUMBER;
 13  BEGIN
 14  	 SELECT cgpa, backlogs, dept_id
 15  	   INTO v_cgpa, v_backlogs, v_dept_id
 16  	   FROM student
 17  	  WHERE student_id = p_student_id;
 18  
 19  	 SELECT min_cgpa, max_backlogs
 20  	   INTO v_min_cgpa, v_max_backlogs
 21  	   FROM drive
 22  	  WHERE drive_id = p_drive_id;
 23  
 24  	 IF v_cgpa < v_min_cgpa THEN
 25  	     RETURN 0;
 26  	 END IF;
 27  
 28  	 IF v_backlogs > v_max_backlogs THEN
 29  	     RETURN 0;
 30  	 END IF;
 31  
 32  	 -- No rows in drive_eligible_dept for this drive => open to all branches.
 33  	 SELECT COUNT(*) INTO v_restricted_count
 34  	   FROM drive_eligible_dept
 35  	  WHERE drive_id = p_drive_id;
 36  
 37  	 IF v_restricted_count > 0 THEN
 38  	     SELECT COUNT(*) INTO v_dept_match_count
 39  	       FROM drive_eligible_dept
 40  	      WHERE drive_id = p_drive_id AND dept_id = v_dept_id;
 41  
 42  	     IF v_dept_match_count = 0 THEN
 43  		 RETURN 0;
 44  	     END IF;
 45  	 END IF;
 46  
 47  	 RETURN 1;
 48  EXCEPTION
 49  	 WHEN NO_DATA_FOUND THEN
 50  	     -- p_student_id or p_drive_id doesn't exist — can't be eligible.
 51  	     RETURN 0;
 52  END fn_check_eligibility;

SQL> -- Rule: a student can hold at most ONE accepted offer (-20003).
SQL> --
SQL> -- WHY A COMPOUND TRIGGER: the check has to count Accepted rows in OFFER, but
SQL> -- this trigger fires on OFFER. A normal row-level trigger that reads its own
SQL> -- table during an UPDATE gets ORA-04091 ("table is mutating"), because the
SQL> -- table is half-way through changing. (Postgres has no such restriction, so
SQL> -- its version is a plain row trigger.) A compound trigger splits the work:
SQL> --   AFTER EACH ROW  - just remember which accepted rows the statement touched
SQL> --   AFTER STATEMENT - the statement has finished, so the table is consistent
SQL> -- 		    and can be read; now lock each student and count.
SQL> -- If any student ends up with more than one Accepted offer the error undoes
SQL> -- the whole statement (Oracle rolls back the failed statement, not the
SQL> -- transaction).
SQL> --
SQL> -- The SELECT ... FOR UPDATE on the student row serializes two concurrent
SQL> -- accepts for the same student, exactly as in the Postgres version: the
SQL> -- second waits for the first to commit, then its count sees both.
SQL> CREATE OR REPLACE TRIGGER trg_offer_before_accept
  2  FOR INSERT OR UPDATE OF status ON offer
  3  COMPOUND TRIGGER
  4  
  5  	 TYPE t_app_ids IS TABLE OF offer.application_id%TYPE;
  6  	 g_accepted_app_ids t_app_ids := t_app_ids();
  7  
  8  	 AFTER EACH ROW IS
  9  	 BEGIN
 10  	     IF :NEW.status = 'Accepted' THEN
 11  		 g_accepted_app_ids.EXTEND;
 12  		 g_accepted_app_ids(g_accepted_app_ids.COUNT) := :NEW.application_id;
 13  	     END IF;
 14  	 END AFTER EACH ROW;
 15  
 16  	 AFTER STATEMENT IS
 17  	     v_student_id student.student_id%TYPE;
 18  	     v_locked	  student.student_id%TYPE;
 19  	     v_accepted   NUMBER;
 20  	 BEGIN
 21  	     FOR i IN 1 .. g_accepted_app_ids.COUNT LOOP
 22  		 SELECT student_id INTO v_student_id
 23  		   FROM application WHERE application_id = g_accepted_app_ids(i);
 24  
 25  		 SELECT student_id INTO v_locked
 26  		   FROM student WHERE student_id = v_student_id FOR UPDATE;
 27  
 28  		 SELECT COUNT(*) INTO v_accepted
 29  		   FROM offer o
 30  		   JOIN application a ON a.application_id = o.application_id
 31  		  WHERE a.student_id = v_student_id
 32  		    AND o.status = 'Accepted';
 33  
 34  		 IF v_accepted > 1 THEN
 35  		     RAISE_APPLICATION_ERROR(-20003,
 36  			 'Student ' || v_student_id || ' has already accepted another offer');
 37  		 END IF;
 38  	     END LOOP;
 39  	 END AFTER STATEMENT;
 40  
 41  END trg_offer_before_accept;
